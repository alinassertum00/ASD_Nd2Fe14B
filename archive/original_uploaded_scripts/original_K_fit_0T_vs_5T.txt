#!/usr/bin/env python3
"""
Fit and compare Nd2Fe14B anisotropy coefficients for:
- separate 0 T Excel files
- one collective 5 T CSV file

This script uses the same fitting model and regularisation settings as the
user-provided fitting code:

F(theta,T) = K1(T) sin^2(theta) + K2(T) sin^4(theta) + K3(T) sin^6(theta)
Ty(theta,T) = -dF/dtheta

Settings:
VOLUME_M3 = 1.1e-24
M_THRESHOLD = 0.95
MINIMUM_ANGLE_DEG = 57.0
LAMBDA_SMOOTH = 0.1
LAMBDA_K3 = 1.0
"""

from pathlib import Path
import csv
import zipfile
import xml.etree.ElementTree as ET
import numpy as np

BASE = Path("/mnt/data")

VOLUME_M3 = 1.1e-24
M_THRESHOLD = 0.95
MINIMUM_ANGLE_DEG = 57.0
LAMBDA_SMOOTH = 0.1
LAMBDA_K3 = 1.0

FILES_0T = {
    25: BASE / "collected_25K_step3_10k(13).xlsx",
    50: BASE / "collected_50K_step3_10k(13).xlsx",
    75: BASE / "collected_75K_step3_10k(14).xlsx",
    100: BASE / "collected_100K_step3_10k(12).xlsx",
    150: BASE / "collected_150K_step3_10k(15).xlsx",
    200: BASE / "collected_200K_step3_10k(13).xlsx",
    250: BASE / "collected_250K_step3_10k(13).xlsx",
    300: BASE / "collected_300K_step3_10k(12).xlsx",
    350: BASE / "collected_350K_step3_10k(12).xlsx",
    400: BASE / "collected_400K_step3_10k(12).xlsx",
    450: BASE / "collected_450K_step3_10k(12).xlsx",
    500: BASE / "collected_500K_step3_10k(12).xlsx",
}

FILE_5T = BASE / "all_temperatures_10K_600K_FIXED(1).csv"


def read_xlsx_sheet1(path: Path):
    namespace = {
        "m": "http://schemas.openxmlformats.org/spreadsheetml/2006/main"
    }

    with zipfile.ZipFile(path) as archive:
        shared_strings = []

        if "xl/sharedStrings.xml" in archive.namelist():
            root = ET.fromstring(archive.read("xl/sharedStrings.xml"))
            for item in root.findall("m:si", namespace):
                shared_strings.append(
                    "".join(
                        node.text or ""
                        for node in item.iter(
                            "{http://schemas.openxmlformats.org/"
                            "spreadsheetml/2006/main}t"
                        )
                    )
                )

        sheet_root = ET.fromstring(
            archive.read("xl/worksheets/sheet1.xml")
        )

        rows = []

        for row in sheet_root.findall(
            ".//m:sheetData/m:row", namespace
        ):
            record = {}

            for cell in row.findall("m:c", namespace):
                reference = cell.attrib["r"]
                column = "".join(
                    character
                    for character in reference
                    if character.isalpha()
                )

                value_node = cell.find("m:v", namespace)
                value = None

                if value_node is not None:
                    raw = value_node.text

                    if cell.attrib.get("t") == "s":
                        value = shared_strings[int(raw)]
                    else:
                        try:
                            value = float(raw)
                        except ValueError:
                            value = raw

                record[column] = value

            rows.append(record)

    return rows


def load_0t_temperature(path: Path):
    rows = read_xlsx_sheet1(path)

    values = []

    for row in rows[1:]:
        try:
            values.append([
                float(row["D"]),
                float(row["E"]),
                float(row["G"]),
            ])
        except (KeyError, TypeError, ValueError):
            continue

    array = np.asarray(values, dtype=float)
    array = array[np.argsort(array[:, 0])]

    return {
        "angle_deg": array[:, 0],
        "M": array[:, 1],
        "Ty_density": array[:, 2] / VOLUME_M3 * 1e-6,
    }


def load_5t_collective(path: Path):
    grouped = {}

    with path.open(newline="", encoding="utf-8") as file:
        reader = csv.DictReader(file)

        for row in reader:
            temperature = int(float(row["Requested_Temperature_K"]))

            grouped.setdefault(temperature, []).append([
                float(row["Angle_deg"]),
                float(row["Output_col_04"]),
                float(row["Output_col_06"]),
            ])

    datasets = {}

    for temperature, values in grouped.items():
        array = np.asarray(values, dtype=float)
        array = array[np.argsort(array[:, 0])]

        datasets[temperature] = {
            "angle_deg": array[:, 0],
            "M": array[:, 1],
            "Ty_density": array[:, 2] / VOLUME_M3 * 1e-6,
        }

    return datasets


def select_contiguous_valid_range(angle_deg, magnetisation):
    maximum = float(np.max(magnetisation))

    invalid = np.where(
        (angle_deg >= 30.0)
        & (magnetisation < M_THRESHOLD * maximum)
    )[0]

    if len(invalid):
        cutoff_index = int(invalid[0] - 1)
    else:
        cutoff_index = len(angle_deg) - 1

    minimum_index = int(
        np.where(angle_deg <= MINIMUM_ANGLE_DEG)[0][-1]
    )

    return max(cutoff_index, minimum_index)


def torque_design(theta):
    return np.column_stack([
        -2.0 * np.sin(theta) * np.cos(theta),
        -4.0 * np.sin(theta) ** 3 * np.cos(theta),
        -6.0 * np.sin(theta) ** 5 * np.cos(theta),
    ])


def fitted_torque(theta, coefficients):
    return torque_design(theta) @ coefficients


def fitted_energy(theta, coefficients):
    K1, K2, K3 = coefficients

    return (
        K1 * np.sin(theta) ** 2
        + K2 * np.sin(theta) ** 4
        + K3 * np.sin(theta) ** 6
    )


def global_fit(raw_by_temperature, temperatures):
    temperatures = np.asarray(temperatures, dtype=float)

    datasets = []

    for temperature in temperatures.astype(int):
        dataset = raw_by_temperature[temperature]

        cutoff_index = select_contiguous_valid_range(
            dataset["angle_deg"],
            dataset["M"],
        )

        theta = np.deg2rad(
            dataset["angle_deg"][:cutoff_index + 1]
        )

        datasets.append({
            **dataset,
            "temperature": temperature,
            "cutoff_index": cutoff_index,
            "cutoff_deg": float(
                dataset["angle_deg"][cutoff_index]
            ),
            "theta_valid": theta,
            "design": torque_design(theta),
            "Ty_valid": dataset["Ty_density"][
                :cutoff_index + 1
            ],
        })

    number_of_temperatures = len(datasets)

    rows = []
    right_hand_side = []

    # Data equations
    for temperature_index, dataset in enumerate(datasets):
        for design_row, measured_value in zip(
            dataset["design"],
            dataset["Ty_valid"],
        ):
            row = np.zeros(3 * number_of_temperatures)

            start = 3 * temperature_index
            row[start:start + 3] = design_row

            rows.append(row)
            right_hand_side.append(measured_value)

    # Temperature smoothness
    for temperature_index in range(
        number_of_temperatures - 1
    ):
        delta_temperature_scaled = (
            temperatures[temperature_index + 1]
            - temperatures[temperature_index]
        ) / 25.0

        for coefficient_index in range(3):
            row = np.zeros(3 * number_of_temperatures)

            row[
                3 * temperature_index + coefficient_index
            ] = (
                -np.sqrt(LAMBDA_SMOOTH)
                / delta_temperature_scaled
            )

            row[
                3 * (temperature_index + 1)
                + coefficient_index
            ] = (
                np.sqrt(LAMBDA_SMOOTH)
                / delta_temperature_scaled
            )

            rows.append(row)
            right_hand_side.append(0.0)

    # K3 stabilisation
    for temperature_index in range(number_of_temperatures):
        row = np.zeros(3 * number_of_temperatures)

        row[3 * temperature_index + 2] = np.sqrt(
            LAMBDA_K3
        )

        rows.append(row)
        right_hand_side.append(0.0)

    matrix = np.asarray(rows)
    vector = np.asarray(right_hand_side)

    solution, _, _, _ = np.linalg.lstsq(
        matrix,
        vector,
        rcond=None,
    )

    coefficients = solution.reshape(
        number_of_temperatures,
        3,
    )

    summary = []

    for dataset, coefficient_set in zip(
        datasets,
        coefficients,
    ):
        predicted = fitted_torque(
            dataset["theta_valid"],
            coefficient_set,
        )

        rmse = float(
            np.sqrt(
                np.mean(
                    (
                        dataset["Ty_valid"]
                        - predicted
                    ) ** 2
                )
            )
        )

        summary.append([
            dataset["temperature"],
            dataset["cutoff_deg"],
            coefficient_set[0],
            coefficient_set[1],
            coefficient_set[2],
            rmse,
        ])

    return np.asarray(summary, dtype=float)


def save_summary(path: Path, summary):
    with path.open("w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)

        writer.writerow([
            "Temperature_K",
            "Cutoff_angle_deg",
            "K1_MJ_m3",
            "K2_MJ_m3",
            "K3_MJ_m3",
            "Torque_RMSE_MJ_m3",
        ])

        writer.writerows(summary)


def main():
    raw_0t = {
        temperature: load_0t_temperature(path)
        for temperature, path in FILES_0T.items()
    }

    raw_5t = load_5t_collective(FILE_5T)

    common_temperatures = sorted(
        set(raw_0t)
        & set(raw_5t)
        & set(range(25, 501))
    )

    summary_0t = global_fit(
        raw_0t,
        common_temperatures,
    )

    summary_5t = global_fit(
        raw_5t,
        common_temperatures,
    )

    save_summary(
        BASE / "anisotropy_fit_summary_0T_reproduced.csv",
        summary_0t,
    )

    save_summary(
        BASE / "anisotropy_fit_summary_5T_reproduced.csv",
        summary_5t,
    )

    print(
        "T(K)   K1_0T   K1_5T   "
        "K2_0T   K2_5T   K3_0T   K3_5T"
    )

    for row_0t, row_5t in zip(summary_0t, summary_5t):
        print(
            f"{int(row_0t[0]):4d}"
            f" {row_0t[2]:8.3f}"
            f" {row_5t[2]:8.3f}"
            f" {row_0t[3]:8.3f}"
            f" {row_5t[3]:8.3f}"
            f" {row_0t[4]:8.3f}"
            f" {row_5t[4]:8.3f}"
        )


if __name__ == "__main__":
    main()
